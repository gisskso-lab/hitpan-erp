## 📋 SQL 12쿼리 (이미 영역 6 v1 박제, v2는 자동화)

### 축 1 (4쿼리)
```sql
-- 1.1 차/대 합계
SELECT SUM(debit_amount), SUM(credit_amount), SUM(debit_amount)-SUM(credit_amount) FROM journal_lines WHERE tenant_id = @tid;

-- 1.2 entry 불균형 수
SELECT COUNT(*) FROM (SELECT entry_id, SUM(debit_amount) d, SUM(credit_amount) c FROM journal_lines WHERE tenant_id = @tid GROUP BY entry_id HAVING ABS(d-c)>0.01) t;

-- 1.3 마이그 vs 운영 분리
SELECT source_type, COUNT(*) FROM journal_entries WHERE tenant_id = @tid GROUP BY source_type;

-- 1.4 미사용 KCODE (accounts에 없는 SC_KCODE)
SELECT COUNT(DISTINCT account_code) FROM journal_lines jl LEFT JOIN accounts a ON a.account_code = jl.account_code AND a.tenant_id = jl.tenant_id WHERE jl.tenant_id = @tid AND a.account_code IS NULL;
```

### 축 2 (3쿼리)
```sql
-- 2.1 매출 총액
SELECT SUM(total_amount) FROM sales_deliveries WHERE tenant_id = @tid AND is_deleted = 0;
SELECT SUM(amount_total+vat_total) FROM tax_invoices WHERE tenant_id = @tid AND direction = '1';

-- 2.2 거래명세서 → 세계 미발급
SELECT COUNT(*) FROM sales_deliveries sd LEFT JOIN tax_invoices ti ON ti.delivery_id = sd.delivery_id WHERE sd.tenant_id = @tid AND ti.invoice_id IS NULL;

-- 2.3 흐름 카운트
SELECT 'q' a, COUNT(*) FROM quotations WHERE tenant_id = @tid UNION ALL SELECT 'so', COUNT(*) FROM sales_orders WHERE tenant_id = @tid UNION ALL SELECT 'sd', COUNT(*) FROM sales_deliveries WHERE tenant_id = @tid UNION ALL SELECT 'ti1', COUNT(*) FROM tax_invoices WHERE tenant_id = @tid AND direction='1';
```

### 축 3 (2쿼리)
```sql
-- 3.1 매입 총액
SELECT SUM(total_amount) FROM purchase_receipts WHERE tenant_id = @tid;
SELECT SUM(amount_total+vat_total) FROM tax_invoices WHERE tenant_id = @tid AND direction='2';

-- 3.2 발주→매입→재고 흐름
SELECT 'po' a, COUNT(*) FROM purchase_orders WHERE tenant_id = @tid UNION ALL SELECT 'pr', COUNT(*) FROM purchase_receipts WHERE tenant_id = @tid;
```

### 축 4 (3쿼리)
```sql
-- 4.1 재고 음수 item
SELECT item_id, SUM(CASE WHEN move_type IN ('in','IN','입고') THEN qty ELSE -qty END) AS 현재고 FROM stock_ledger WHERE tenant_id = @tid GROUP BY item_id HAVING 현재고 < 0 LIMIT 50;

-- 4.2 거래명세서 ↔ 재고원장 미연결
SELECT COUNT(*) FROM sales_deliveries sd LEFT JOIN stock_ledger sl ON sl.source_id LIKE CONCAT('%', sd.delivery_id, '%') WHERE sd.tenant_id = @tid AND sl.ledger_id IS NULL;

-- 4.3 재고원장 라인 수
SELECT COUNT(*) FROM stock_ledger WHERE tenant_id = @tid;
```

## 📋 PowerShell 스크립트 골격

```powershell
param([string]$TenantId)
$DBHost = "localhost"
$DBUser = "hitpan"
$DBPass = "Hitpan2025!"
$DBName = "hitpan_erp"

function Run-MySQL($sql) {
    & mysql -h $DBHost -u $DBUser -p"$DBPass" $DBName -e $sql --skip-column-names
}

Write-Host "끝숫자 4축 자동 검증 시작 ($(Get-Date))" -ForegroundColor Cyan
Write-Host "Tenant: $TenantId`n"

# 축 1
$debit = Run-MySQL "SELECT SUM(debit_amount) FROM journal_lines WHERE tenant_id='$TenantId';"
$credit = Run-MySQL "SELECT SUM(credit_amount) FROM journal_lines WHERE tenant_id='$TenantId';"
$diff = [decimal]$debit - [decimal]$credit
$unbalanced = Run-MySQL "SELECT COUNT(*) FROM (SELECT entry_id, SUM(debit_amount) d, SUM(credit_amount) c FROM journal_lines WHERE tenant_id='$TenantId' GROUP BY entry_id HAVING ABS(d-c)>0.01) t;"
$entryCount = Run-MySQL "SELECT COUNT(*) FROM journal_entries WHERE tenant_id='$TenantId';"

$axis1Score = if ($diff -eq 0 -and $unbalanced -eq 0) { 100 } else { [math]::Max(0, 100 - [int]($unbalanced * 100 / $entryCount)) }

Write-Host "축 1: 회계 정합성" -ForegroundColor Yellow
Write-Host "  차변 합계: $($debit -as [decimal] | ForEach-Object { $_.ToString('N0') })원"
Write-Host "  대변 합계: $($credit -as [decimal] | ForEach-Object { $_.ToString('N0') })원"
Write-Host "  차-대 차이: $($diff.ToString('N0'))원 $(if($diff -eq 0){'✅'}else{'🚨'})"
Write-Host "  불균형 entry: $unbalanced / $entryCount $(if($unbalanced -eq 0){'✅'}else{'🚨'})"
Write-Host "  → $axis1Score 점`n"

# 축 2·3·4 동형 ...

# 종합 점수
$totalScore = ($axis1Score + $axis2Score + $axis3Score + $axis4Score) / 4
Write-Host "━━━━━━━━━━━━━━━━━━━━" -ForegroundColor Cyan
Write-Host "종합 점수: $totalScore / 100"
Write-Host "베타 게이트: $(if($totalScore -ge 90){'✅ 통과'}else{'🚨 미통과'})"
Write-Host "━━━━━━━━━━━━━━━━━━━━`n"
```
