// 2. 테이블별 분리 tx (각각 30초 내 commit)
async Task<int> MigrateTableWithCheckpoint(
    string tableName,
    Func<IDbTransaction, Task<int>> migrateAction,
    CancellationToken ct)
{
    // 체크포인트 확인 (재시작 시 이미 done인 테이블 스킵)
    var existing = await _db.QueryFirstOrDefaultAsync<string>(
        "SELECT status FROM migration_checkpoints WHERE job_id=@J AND table_name=@T",
        new { J = jobId, T = tableName });
    if (existing == "done") return 0; // 멱등 스킵

    // pending → running
    await UpsertCheckpoint(tableName, "running", ct);

    using var tx = _db.BeginTransaction();
    try
    {
        var inserted = await migrateAction(tx);
        tx.Commit();
        await UpsertCheckpoint(tableName, "done", inserted, ct);
        return inserted;
    }
    catch (Exception ex)
    {
        try { tx.Rollback(); } catch { }
        await UpsertCheckpoint(tableName, "failed", error: ex.Message, ct);
        await InsertMigrationError(tableName, ex, ct); // raw_data AES INSERT (P0-3)
        throw;
    }
}

// 3. PYOJUN.MDB 4 테이블
using (var oleConn = OpenOleDb(pyojunPath))
{
    result.Partners = await MigrateTableWithCheckpoint("partners",
        tx => MigratePartnersAsync(oleConn, tenantId, now, partnerMap, tx, ct), ct);
    result.Items = await MigrateTableWithCheckpoint("items",
        tx => MigrateItemsAsync(oleConn, tenantId, now, itemMap, tx, ct), ct);
    // ... 14개 추가
}

- 재시작 시 done 테이블 스킵
- undo log < 10초 (단일 테이블당 평균 280 rows/sec)

## P0-3. 헌법 #5 — migration_errors.raw_data AES INSERT 구현

### 위반 위치
- `src/HitPan.Application/Services/MdbMigrationService.cs` — `migration_errors` INSERT 로직 0건
- DB: `migration_errors.raw_data longtext utf8mb4_bin` — 평문 가능 (P0)

### 봉합 안

**MdbMigrationService에 추가:**
```csharp
private async Task InsertMigrationErrorAsync(
    string jobId, string tenantId, string tableName,
    string rowPkValue, int? rowOffset,
    string errorType, string errorSeverity, string errorCode,
    string errorMessage, string errorDetail,
    object? rawRowData, CancellationToken ct)
{
    // raw_data: AES 암호화 후 longtext에 Base64 저장
    string? rawDataEncrypted = null;
    if (rawRowData != null)
    {
        var json = System.Text.Json.JsonSerializer.Serialize(rawRowData);
        var bytes = _crypto.EncryptToBytes(json); // IBinaryCryptoService
        rawDataEncrypted = Convert.ToBase64String(bytes);
    }

    const string sql = """
        INSERT INTO migration_errors
          (error_id, job_id, tenant_id, mdb_file, table_name,
           row_pk_value, row_offset,
           error_type, error_severity, error_code,
           error_message, error_detail,
           raw_data, occurred_at, created_at)
        VALUES
          (UUID(), @JobId, @TenantId, @MdbFile, @TableName,
           @RowPk, @RowOffset,
           @ErrorType, @ErrorSeverity, @ErrorCode,
           @ErrorMessage, @ErrorDetail,
           @RawData, NOW(6), NOW(6))
        """;

    await _db.ExecuteAsync(new CommandDefinition(sql, new
    {
        JobId = jobId, TenantId = tenantId, MdbFile = "PYOJUN.MDB",
        TableName = tableName, RowPk = rowPkValue, RowOffset = rowOffset,
        ErrorType = errorType, ErrorSeverity = errorSeverity, ErrorCode = errorCode,
        ErrorMessage = errorMessage.Length > 1000 ? errorMessage[..1000] : errorMessage,
        ErrorDetail = errorDetail,
        RawData = rawDataEncrypted
    }, commandTimeout: 30, cancellationToken: ct));
}
```

### 효과
- 마이그 행 실패 시 원본 데이터 AES 암호화 보관
- 헌법 #5 (raw_data 평문 노출 0) 달성

- stock_ledger 116,420행 마이그 — 한 번에 BulkCopy

### 봉합 안 (W3 청크화)

**MdbMigrationService.cs:L757 MigrateStockLedgerAsync 변경:**
```csharp
private async Task<int> MigrateStockLedgerAsync(...)
{
    var rows = ReadMdbTable(oleConn,
        "SELECT * FROM DOCFB ORDER BY IJ_DT, IJ_IO, IJ_SEQ, IJ_BUY, IJ_SUN"); // P0-4

    const int CHUNK = 5000; // W3 권고
    var totalInserted = 0;
    var lastPk = ""; // 체크포인트용

    for (int offset = 0; offset < rows.Rows.Count; offset += CHUNK)
    {
        var chunk = rows.Rows.Cast<DataRow>().Skip(offset).Take(CHUNK).ToList();
        var chunkRows = new List<object?[]>();
        foreach (var row in chunk)
        {
            // 변환 로직 (기존 L:770-805)
            // ...
            chunkRows.Add(/* 변환된 행 */);
            lastPk = $"{GetStr(row,"IJ_DT")}-{GetShort(row,"IJ_SEQ")}";
        }

        var inserted = await BulkInsertAsync("stock_ledger", columns, chunkRows, tx, ct);
        totalInserted += inserted;

        // 체크포인트 갱신 (last_pk_value JSON)
        await UpdateCheckpointAsync(tableName: "stock_ledger",
            processedCount: totalInserted,
            lastPkValue: System.Text.Json.JsonSerializer.Serialize(new { IJ_DT_SEQ = lastPk }),
            chunkSize: CHUNK,
            ct);
    }
    return totalInserted;
}
```

### 효과
- 116K행을 5K씩 24청크로 분할 → 각 청크 < 7초
- last_pk_value JSON으로 재시작 가능
- UNIQUE 인덱스 + ON DUPLICATE KEY → 멱등 100%
