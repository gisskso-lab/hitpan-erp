## 🎯 4축 자동 검증 SQL 도구 (PM 1차 안)

### 축 1 — 회계 정합성
```sql
-- 1.1 차/대변 절대 일치
SELECT
  '회계 차변합계' axis,
  ROUND(SUM(debit_amount), 0) val
FROM journal_lines
WHERE tenant_id = @tenant_id
UNION ALL SELECT '회계 대변합계', ROUND(SUM(credit_amount), 0) FROM journal_lines WHERE tenant_id = @tenant_id
UNION ALL SELECT '회계 차-대 차이', ROUND(SUM(debit_amount)-SUM(credit_amount), 0) FROM journal_lines WHERE tenant_id = @tenant_id
;

-- 1.2 entry별 차/대 불균형 수
SELECT COUNT(*) AS 불균형_entry수
FROM (
  SELECT entry_id, SUM(debit_amount) d, SUM(credit_amount) c
  FROM journal_lines WHERE tenant_id = @tenant_id
  GROUP BY entry_id
  HAVING ABS(d-c) > 0.01
) t;

-- 1.3 마이그 데이터 vs 운영 데이터 분리 정합성
SELECT
  CASE WHEN source_type = 'migration' THEN '마이그' ELSE '운영' END AS data_source,
  COUNT(*) AS entry_수,
  ROUND(SUM(amount), 0) AS 총액
FROM journal_entries WHERE tenant_id = @tenant_id GROUP BY 1;
```

### 축 2 — 영업 (매출) 정합성
```sql
-- 2.1 거래명세서 vs 세금계산서 매출 총액 일치
SELECT
  'sales_deliveries 매출' axis, ROUND(SUM(total_amount), 0) val
FROM sales_deliveries WHERE tenant_id = @tenant_id AND is_deleted = 0
UNION ALL SELECT 'tax_invoices 매출(direction=1)', ROUND(SUM(amount_total+vat_total), 0)
FROM tax_invoices WHERE tenant_id = @tenant_id AND direction = '1'
;

-- 2.2 거래명세서 → 세금계산서 연결 누락 (delivery_id 매칭)
SELECT
  COUNT(*) AS 거래명세서_세계_미발급
FROM sales_deliveries sd
LEFT JOIN tax_invoices ti ON ti.delivery_id = sd.delivery_id AND ti.tenant_id = sd.tenant_id
WHERE sd.tenant_id = @tenant_id AND ti.invoice_id IS NULL;

-- 2.3 견적서 → 수주 → 거래명세서 → 세금계산서 흐름 끊김
SELECT
  '견적서수' a, COUNT(*) b FROM quotations WHERE tenant_id = @tenant_id
UNION ALL SELECT '수주수', COUNT(*) FROM sales_orders WHERE tenant_id = @tenant_id
UNION ALL SELECT '거래명세수', COUNT(*) FROM sales_deliveries WHERE tenant_id = @tenant_id
UNION ALL SELECT '세계수', COUNT(*) FROM tax_invoices WHERE tenant_id = @tenant_id AND direction = '1'
;
```

### 축 3 — 구매 정합성
```sql
-- 3.1 거래명세서 vs 세금계산서 매입 총액 일치
SELECT
  'purchase_receipts 매입' axis, ROUND(SUM(total_amount), 0) val
FROM purchase_receipts WHERE tenant_id = @tenant_id
UNION ALL SELECT 'tax_invoices 매입(direction=2)', ROUND(SUM(amount_total+vat_total), 0)
FROM tax_invoices WHERE tenant_id = @tenant_id AND direction = '2'
;

-- 3.2 발주서 → 매입 → 재고 흐름
SELECT
  '발주수' a, COUNT(*) b FROM purchase_orders WHERE tenant_id = @tenant_id
UNION ALL SELECT '매입수', COUNT(*) FROM purchase_receipts WHERE tenant_id = @tenant_id
UNION ALL SELECT '매입세계수', COUNT(*) FROM tax_invoices WHERE tenant_id = @tenant_id AND direction = '2'
;
```

### 축 4 — 재고 정합성
```sql
-- 4.1 재고원장 입출고 균형 (item별)
SELECT
  item_id,
  SUM(CASE WHEN move_type IN ('in','IN','입고') THEN qty ELSE 0 END) AS 입고합,
  SUM(CASE WHEN move_type IN ('out','OUT','출고') THEN qty ELSE 0 END) AS 출고합,
  SUM(CASE WHEN move_type IN ('in','IN','입고') THEN qty ELSE 0 END)
    - SUM(CASE WHEN move_type IN ('out','OUT','출고') THEN qty ELSE 0 END) AS 현재고
FROM stock_ledger WHERE tenant_id = @tenant_id
GROUP BY item_id
HAVING 현재고 < 0  -- 음수 재고 = 워크플로우 끊김
LIMIT 50;

-- 4.2 재고원장 vs 거래명세서 연결
SELECT
  COUNT(*) AS 거래명세서_재고원장_미연결
FROM sales_deliveries sd
LEFT JOIN stock_ledger sl ON sl.source_id LIKE CONCAT('%', sd.delivery_id, '%')
WHERE sd.tenant_id = @tenant_id AND sl.ledger_id IS NULL;
```
