       LEFT(error_message, 200) AS msg_preview,
       LEFT(error_detail, 200)  AS detail_preview
  FROM migration_errors
 WHERE created_at >= NOW() - INTERVAL 1 HOUR
 ORDER BY created_at DESC
 LIMIT 20;

-- 주민번호 패턴 검사 (마스킹 안 된 평문)

  FROM migration_errors
 WHERE created_at >= NOW() - INTERVAL 1 HOUR
   AND (error_message REGEXP '[0-9]{6}-[1-4][0-9]{6}'
        OR error_detail REGEXP '[0-9]{6}-[1-4][0-9]{6}');

  FROM migration_errors

        OR error_detail REGEXP '[0-9]{3}-[0-9]{2}-[0-9]{5}');
