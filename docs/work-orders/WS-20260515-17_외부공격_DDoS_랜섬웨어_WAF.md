```csharp
public sealed class FileIntegrityWorker : BackgroundService {
    private readonly Dictionary<string, string> _baseline = new();

    protected override async Task ExecuteAsync(CancellationToken ct) {
        // 시작 시 핵심 파일 해시 베이스라인
        await BuildBaselineAsync();

        while (!ct.IsCancellationRequested) {
            await CheckIntegrityAsync(ct);
            await Task.Delay(TimeSpan.FromMinutes(5), ct);
        }
    }

    private async Task CheckIntegrityAsync(CancellationToken ct) {
        foreach (var (path, baselineHash) in _baseline) {
            if (!File.Exists(path)) {
                _logger.LogError("[FileIntegrity] 파일 삭제 감지: {Path}", path);
                await TriggerEmergencyBackupAsync(ct);
                continue;
            }
            var currentHash = await ComputeHashAsync(path, ct);
            if (currentHash != baselineHash) {
                _logger.LogError("[FileIntegrity] 파일 변조 감지: {Path}", path);
                await TriggerEmergencyBackupAsync(ct);
            }
        }
    }
}
```

### 6-2. 로컬 Bot 차단 (User-Agent + 행동 패턴)

```csharp
public sealed class BotDetectionMiddleware {
    private static readonly string[] SuspiciousUAs = {
        "sqlmap", "nikto", "nmap", "masscan", "zgrab",
        "python-requests", "curl/", "wget/",  // 비ERP 환경
    };

    public async Task InvokeAsync(HttpContext context) {
        var ua = context.Request.Headers.UserAgent.ToString();
        if (SuspiciousUAs.Any(s => ua.Contains(s, StringComparison.OrdinalIgnoreCase))) {
            _logger.LogWarning("[Bot] 의심 UA 차단: {UA} from {IP}",
                ua, context.Connection.RemoteIpAddress);
            context.Response.StatusCode = 403;
            await context.Response.WriteAsync("Forbidden.");
            return;
        }
        await _next(context);
    }
}
```
