## 1. 목적

`DOCF4` (MSSQL 정답) ↔ `tax_invoices` (MariaDB 현행) 스키마 정합. 5/14 마이그 12/13 PASS 잔여 1건이 세금계산서. DDL/마이그 코드 동시 봉합으로 13/13 PASS 달성.

## 2. 정답 스키마 (MSSQL DOCF4)

| 컬럼 | 타입 | 의미 |
|---|---|---|
| TX_IO | char(1) | 매출(S)/매입(B) 구분 |
| TX_NO | char(8) | 세금계산서 번호 |
| TX_PDT | char(8) | 발행일 (YYYYMMDD) |
| TX_BUY | int | 거래처 코드 |
| TX_PUM1~4 | varchar(100) | 품목명 4개 인라인 |
| TX_SU1~4 | money | 수량 4개 |
| TX_DAN1~4 | money | 단가 4개 |
| TX_KUM1~4 | money | 공급가 4개 |
| TX_VAT1~4 | money | 부가세 4개 |
| TX_GU, TX_GU1 | char(1) | 구분 |
| TX_SEQ | smallint | 순번 |
| TX_OLD | money | 미수액 |
| TX_Y1, TX_Y2 | smallint | 연도 |
| TX_SENDDT | char(8) | 발송일 |
| TX_READDT | char(8) | 확인일 |
| TX_REPORTDT | char(8) | 신고일 |
| TX_REM, TX_REM1 | varchar(100) | 비고 |

**PK 정답:** `TX_IO + TX_NO` (2 컬럼 복합)

## 3. 현행 tax_invoices 문제점

```sql
delivery_id   varchar(36)  NOT NULL UNIQUE  -- ❗ 거래명세서 없는 세금계산서 INSERT 불가
invoice_no    varchar(32)
issued_at     datetime(6)
amount_total, vat_total
```

문제:
1. **`delivery_id NOT NULL UNIQUE`** — 레거시 마이그 시 거래명세서가 1:1로 없으면 봉합 불가
2. TX_IO (매출/매입 구분) 컬럼 없음
3. TX_SENDDT / TX_READDT / TX_REPORTDT 별도 추적 컬럼 없음
4. 4 품목 인라인 분해 대상 `tax_invoice_items` 테이블 부재
5. PK 정답이 `(TX_IO, TX_NO)` 복합인데 MariaDB는 GUID 단일 PK + delivery_id UNIQUE

## 4. 봉합 액션 (3 단계)

### 단계 1: DDL ALTER (DB매니저)

```sql
-- 4-1. delivery_id NOT NULL 해제 (레거시 마이그 호환)
ALTER TABLE tax_invoices
  MODIFY delivery_id varchar(36) NULL,
  DROP INDEX `delivery_id`;

-- 4-2. TX_* 컬럼 추가
ALTER TABLE tax_invoices
  ADD COLUMN direction CHAR(1) NULL COMMENT 'S=매출, B=매입 (TX_IO)',
  ADD COLUMN tax_no CHAR(8) NULL COMMENT 'TX_NO 세금계산서 번호',
  ADD COLUMN issue_date_yyyymmdd CHAR(8) NULL COMMENT 'TX_PDT',
  ADD COLUMN partner_code INT NULL COMMENT 'TX_BUY 레거시 거래처 코드',
  ADD COLUMN seq_no SMALLINT NULL COMMENT 'TX_SEQ',
  ADD COLUMN sent_at_yyyymmdd CHAR(8) NULL COMMENT 'TX_SENDDT',
  ADD COLUMN read_at_yyyymmdd CHAR(8) NULL COMMENT 'TX_READDT',
  ADD COLUMN reported_at_yyyymmdd CHAR(8) NULL COMMENT 'TX_REPORTDT',
  ADD COLUMN remark1 VARCHAR(100) NULL COMMENT 'TX_REM',
  ADD COLUMN remark2 VARCHAR(100) NULL COMMENT 'TX_REM1',
  ADD COLUMN source_type VARCHAR(30) NULL,
  ADD COLUMN source_id VARCHAR(80) NULL,
  ADD UNIQUE KEY uq_tax_invoices_io_no (tenant_id, direction, tax_no),
  ADD UNIQUE KEY uq_tax_invoices_source (tenant_id, source_type, source_id);

-- 4-3. tax_invoice_items 신규 (4 품목 인라인 분해 결과 저장)
CREATE TABLE IF NOT EXISTS tax_invoice_items (
  tax_invoice_item_id VARCHAR(36) PRIMARY KEY,
  invoice_id VARCHAR(36) NOT NULL,
  tenant_id VARCHAR(36) NOT NULL,
  line_no SMALLINT NOT NULL COMMENT '1~4 (TX_PUM1~4 분해)',
  item_name VARCHAR(100) NOT NULL,
  quantity DECIMAL(15,2) NOT NULL DEFAULT 0,
  unit_price DECIMAL(15,2) NOT NULL DEFAULT 0,
  supply_amount DECIMAL(15,2) NOT NULL DEFAULT 0,
  vat_amount DECIMAL(15,2) NOT NULL DEFAULT 0,
  created_at DATETIME(6) NOT NULL DEFAULT CURRENT_TIMESTAMP(6),
  UNIQUE KEY uq_tax_invoice_item_line (invoice_id, line_no),
  KEY idx_tax_invoice_items_tenant (tenant_id),
  CONSTRAINT fk_tax_invoice_items_invoice FOREIGN KEY (invoice_id) REFERENCES tax_invoices(invoice_id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
```

### 단계 2: 마이그 코드 (백엔드매니저)

`MdbMigrationService.cs` — `MigrateTaxInvoicesAsync` 신규 메서드 (또는 기존 메서드 정공법 교체):

```csharp
private async Task<int> MigrateTaxInvoicesAsync(
    OleDbConnection oleConn, string tenantId, DateTime now,
    Dictionary<int, string> partnerMap,
    IDbTransaction tx, CancellationToken ct)
{
    // PK 정답 순서대로 ORDER BY
    var dt = ReadMdbTable(oleConn,
        "SELECT * FROM DOCF4 ORDER BY TX_IO, TX_NO");
    if (dt.Rows.Count == 0) return 0;

    const string headerSql = """
        INSERT INTO tax_invoices
          (invoice_id, tenant_id, direction, tax_no, issue_date_yyyymmdd,
           partner_code, seq_no, sent_at_yyyymmdd, read_at_yyyymmdd, reported_at_yyyymmdd,
           remark1, remark2,
           invoice_no, issued_at, issued_by, amount_total, vat_total, status,
           etax_status, source_type, source_id, migrated_source_hash,
           created_at, updated_at)
        VALUES
          (@Id, @TenantId, @Direction, @TaxNo, @IssueDate,
           @PartnerCode, @Seq, @SentDt, @ReadDt, @ReportDt,
           @Rem1, @Rem2,
           @TaxNo, @IssuedAt, @MigBy, @Amount, @Vat, 'issued',
           'migrated', 'migration', @SourceId, @Hash,
           @Now, @Now)
        ON DUPLICATE KEY UPDATE
          amount_total = VALUES(amount_total),
          vat_total = VALUES(vat_total),
          updated_at = VALUES(updated_at)
        """;

    const string lineSql = """
        INSERT IGNORE INTO tax_invoice_items
          (tax_invoice_item_id, invoice_id, tenant_id, line_no,
           item_name, quantity, unit_price, supply_amount, vat_amount, created_at)
        VALUES
          (@LineId, @InvoiceId, @TenantId, @LineNo,
           @ItemName, @Qty, @UnitPrice, @Supply, @Vat, @Now)
        """;

    int count = 0;
    foreach (DataRow r in dt.Rows)
    {
        var io = GetStr(r, "TX_IO");
        var txNo = GetStr(r, "TX_NO");
        if (string.IsNullOrWhiteSpace(txNo)) continue;

        var sourceId = $"mig-{io}-{txNo}";
        var invoiceId = Guid.NewGuid().ToString();
        decimal supply = 0, vat = 0;

        for (int i = 1; i <= 4; i++)
        {
            supply += GetDec(r, $"TX_KUM{i}");
            vat += GetDec(r, $"TX_VAT{i}");
        }

        await Db.ExecuteAsync(new CommandDefinition(headerSql, new
        {
            Id = invoiceId,
            TenantId = tenantId,
            Direction = io,
            TaxNo = txNo,
            IssueDate = GetStr(r, "TX_PDT"),
            PartnerCode = GetInt(r, "TX_BUY"),
            Seq = GetInt(r, "TX_SEQ"),
            SentDt = GetStr(r, "TX_SENDDT"),
            ReadDt = GetStr(r, "TX_READDT"),
            ReportDt = GetStr(r, "TX_REPORTDT"),
            Rem1 = GetStr(r, "TX_REM"),
            Rem2 = GetStr(r, "TX_REM1"),
            IssuedAt = ParseLegacyDate(GetStr(r, "TX_PDT")) ?? now,
            MigBy = "migration",
            Amount = supply,
            Vat = vat,
            SourceId = sourceId,
            Hash = ComputeSourceHash($"tax_invoices:{sourceId}:{supply}:{vat}"),
            Now = now,
        }, transaction: tx, cancellationToken: ct)).ConfigureAwait(false);

        // 4 품목 인라인 분해
        for (int i = 1; i <= 4; i++)
        {
            var pum = GetStr(r, $"TX_PUM{i}");
            if (string.IsNullOrWhiteSpace(pum)) continue;
            await Db.ExecuteAsync(new CommandDefinition(lineSql, new
            {
                LineId = Guid.NewGuid().ToString(),
                InvoiceId = invoiceId,
                TenantId = tenantId,
                LineNo = (short)i,
                ItemName = pum,
                Qty = GetDec(r, $"TX_SU{i}"),
                UnitPrice = GetDec(r, $"TX_DAN{i}"),
                Supply = GetDec(r, $"TX_KUM{i}"),
                Vat = GetDec(r, $"TX_VAT{i}"),
                Now = now,
            }, transaction: tx, cancellationToken: ct)).ConfigureAwait(false);
        }
        count++;
    }

    _logger.LogInformation("[MDB마이그레이션] 세금계산서(DOCF4→tax_invoices) {Count}건 이관 완료", count);
    return count;
}
```

### 단계 3: 검증 (DB매니저)

-- 13/13 PASS 카드 갱신
SELECT COUNT(*) FROM tax_invoices WHERE tenant_id = @tenant;

SELECT invoice_id, COUNT(*) AS line_count
FROM tax_invoice_items
WHERE tenant_id = @tenant
GROUP BY invoice_id
HAVING line_count > 4;  -- 0건이어야 정상

-- 멱등 키 중복 0건 확인
SELECT COUNT(*) - COUNT(DISTINCT source_id) AS dup

## 6. 영향 범위

- **신규 컬럼만 추가** — 기존 코드 (issue_at, invoice_no 등)는 그대로 동작
- **delivery_id NOT NULL 해제** = 마이그 row만 NULL. 신규 운영 row는 코드에서 채워야 함 (FE/BE 변경 불필요, 기존 INSERT 동작 영향 없음)
- **tax_invoice_items 신규** = 기존 코드 영향 없음 (CASCADE FK만 추가)
