```csharp
private async Task<string> EnsureLegacyFallbackItemAsync(
    string tenantId, DateTime now, IDbTransaction tx, CancellationToken ct)
{
    const string code = "LEGACY-UNKNOWN-ITEM";
    var existing = await Db.ExecuteScalarAsync<string?>(new CommandDefinition(
        "SELECT item_id FROM items WHERE tenant_id = @TenantId AND item_code = @Code LIMIT 1",
        new { TenantId = tenantId, Code = code },
        transaction: tx, cancellationToken: ct)).ConfigureAwait(false);

    if (!string.IsNullOrEmpty(existing)) return existing;

    var itemId = Guid.NewGuid().ToString();
    const string sql = """
        INSERT INTO items (item_id, tenant_id, item_code, item_name, item_type, unit,
                          purchase_price, sale_price, ..., is_active, is_deleted, created_at, updated_at)
        VALUES (@Id, @TenantId, 'LEGACY-UNKNOWN-ITEM', '레거시 미등록 품목', 'product', 'EA',
                0, 0, ..., 1, 0, @Now, @Now)
        """;
    await Db.ExecuteAsync(new CommandDefinition(sql, ..., transaction: tx, cancellationToken: ct));
    return itemId;
}
```
