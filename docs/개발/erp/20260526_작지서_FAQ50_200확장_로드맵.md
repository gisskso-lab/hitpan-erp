## 📋 FAQ 표준 포맷

### DB 스키마 (DB 매니저 가도, 작H 발행 시)
```sql
CREATE TABLE faq (
  faq_id INT UNSIGNED NOT NULL AUTO_INCREMENT,
  category ENUM('install', 'master', 'workflow', 'tax', 'subscription') NOT NULL,
  question VARCHAR(500) NOT NULL,
  answer TEXT NOT NULL,
  keywords VARCHAR(500) NOT NULL COMMENT '검색 키워드 콤마 구분',
  view_count INT UNSIGNED NOT NULL DEFAULT 0,
  helpful_count INT UNSIGNED NOT NULL DEFAULT 0,
  author VARCHAR(100) NOT NULL COMMENT '작성자 (마케팅/기술영업/ERP/법무)',
  is_published TINYINT(1) NOT NULL DEFAULT 1,
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (faq_id),
  INDEX idx_category_published (category, is_published),
  FULLTEXT idx_search (question, answer, keywords)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
```
