### 1. 사장님 2중 암호화 + TPM 2.0 봉인

#### 저장 단계 (인증서 등록 시)
```csharp
public class DoubleEncryptionService
{
    private readonly ITpmKeyService _tpm;

    public async Task RegisterCertAsync(byte[] pfxBytes, string userPin)
    {
        // 1차: 사용자 PIN으로 PFX 복호화 → 원본 추출
        var rawPfx = DecryptPfx(pfxBytes, userPin);

        // 2차: ERP 마스터 키 생성 (PC별 고유)
        var erpMasterKey = GenerateMasterKey();

        // 3차: TPM 2.0 봉인 (NCRYPT_PLATFORM_KEY_FLAG)
        var sealedKey = _tpm.SealKey(erpMasterKey);

        // 4차: ERP 마스터 키로 PFX 2차 암호화
        var doubleEncryptedPfx = EncryptWithMasterKey(rawPfx, erpMasterKey);

        // 5차: DPAPI 격리 저장 (%LOCALAPPDATA%\HitPan\Secure\)
        await SaveSecureAsync(doubleEncryptedPfx, sealedKey);

        // 6차: SecureString + 메모리 즉시 폐기
        Array.Clear(rawPfx, 0, rawPfx.Length);
        Array.Clear(erpMasterKey, 0, erpMasterKey.Length);
        GC.Collect();
    }
}
```

#### 사용 단계 (발급 시)
```csharp
public async Task<X509Certificate2> LoadCertAsync(string userPin)
{
    // 1. 원격제어 감지 (TeamViewer·AnyDesk·RustDesk·Chrome Remote)
    if (RemoteControlDetector.IsActive())
        throw new SecurityException("원격제어 감지됨. 발급 차단");

    // 2. 가상 키패드로 PIN 입력 (Anti-Keylogger)
    if (string.IsNullOrEmpty(userPin))
        userPin = await _virtualKeypad.GetPinAsync();

    // 3. DPAPI → 봉인 키 복호화
    var (encryptedPfx, sealedKey) = await LoadSecureAsync();

    // 4. TPM → ERP 마스터 키 추출
    var erpMasterKey = _tpm.UnsealKey(sealedKey);

    // 5. ERP 키로 PFX 2차 복호화
    var encryptedRawPfx = DecryptWithMasterKey(encryptedPfx, erpMasterKey);

    // 6. 사용자 PIN으로 1차 복호화 → 원본 인증서
    var cert = new X509Certificate2(encryptedRawPfx, userPin);

    // 7. 메모리 즉시 폐기 (SecureString)
    Array.Clear(erpMasterKey, 0, erpMasterKey.Length);
    Array.Clear(encryptedRawPfx, 0, encryptedRawPfx.Length);

    return cert;
}
```

        "TeamViewer", "AnyDesk", "RustDesk",
        "rdpclip", "chrome_remote_desktop_host",
        "ScreenConnect", "LogMeIn"

    public static bool IsActive()
    {
        return Process.GetProcesses()
            .Any(p => BLOCKED_PROCESSES.Contains(p.ProcessName));
    }

### 홈택스 API 호출
```csharp
public class DirectHometaxProvider : ITaxInvoiceProvider
{
    public async Task<IssueResult> IssueAsync(TaxInvoice invoice, X509Certificate2 cert)
    {
        // 1. 휴폐업 조회
        var status = await CheckBusinessStatusAsync(invoice.SupplierBizNo);
        if (!status.IsActive)
            return IssueResult.Fail("휴폐업 사업자");

        // 2. XML 생성 (KS X ISO/IEC 19845, 36 필수필드)
        var xml = _xmlBuilder.Build(invoice);

        // 3. SignedXml 서명 (NPKI RSA-SHA256)
        var signed = SignXml(xml, cert);

        // 4. 홈택스 API POST (HttpClient + Polly 재시도)
        var response = await _http.PostAsync(HOMETAX_URL, signed)
            .WithRetry(3, exponentialBackoff: true);

        // 5. 멱등키 저장 (중복 발급 차단)
        await SaveIdempotencyKeyAsync(invoice.Id, response.ApprovalNo);

        // 6. 반송 폴링 큐 등록 (5분~24h 비동기)
        await EnqueueBouncePollingAsync(response.ApprovalNo);

        return IssueResult.Success(response.ApprovalNo);
    }
}
```
