##### 가설 ① 대응: Content-Disposition 변경
```csharp
// DocumentController.cs
[HttpGet("pdf")]
public async Task<IActionResult> GetPdf([FromQuery] string token)
{
    var pdfBytes = await _pdfService.GenerateAsync(token);
    return File(pdfBytes, "application/pdf", "quotation.pdf");
    // ↓ 모바일 인라인 표시로 변경
    Response.Headers.Add("Content-Disposition", "inline; filename=quotation.pdf");
    return File(pdfBytes, "application/pdf");
}
```

##### 가설 ④ 대응: window.open → location.href
```javascript
// hitpan.js
window.hitpan.openFile = function(url) {
    if (window.navigator.userAgent.match(/KAKAOTALK|NAVER/i)) {
        // 인앱 브라우저: location.href 사용
        window.location.href = url;
    } else {
        window.open(url, '_blank');
    }
};
```
