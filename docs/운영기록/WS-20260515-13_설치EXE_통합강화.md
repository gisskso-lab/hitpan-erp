## 5. 실패 시 자동 롤백

```csharp
public sealed class InstallTransaction {
    private readonly Stack<Func<Task>> _rollbackActions = new();

    public async Task RegisterAsync(Func<Task> install, Func<Task> rollback) {
        await install();
        _rollbackActions.Push(rollback);
    }

    public async Task RollbackAsync() {
        while (_rollbackActions.TryPop(out var action)) {
            try { await action(); } catch { /* best-effort */ }
        }
    }
}

// 사용
var tx = new InstallTransaction();
try {
    await tx.RegisterAsync(InstallDotNet8Async, UninstallDotNet8Async);
    await tx.RegisterAsync(InstallMariaDbAsync, UninstallMariaDbAsync);
    // ...
} catch {
    await tx.RollbackAsync();
    throw;
}
```
