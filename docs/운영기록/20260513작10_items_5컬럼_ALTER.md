## 3. ALTER 실행 SQL

```sql
ALTER TABLE items
    ADD COLUMN IF NOT EXISTS spec_detail VARCHAR(80) NULL
        COMMENT '상세 규격 (S_SPEC)',
    ADD COLUMN IF NOT EXISTS unit_secondary VARCHAR(10) NULL
        COMMENT '2차 단위 (S_UNIT2)',
    ADD COLUMN IF NOT EXISTS safety_stock DECIMAL(15,3) DEFAULT 0
        COMMENT '안전 재고 (S_SAFE)',
    ADD COLUMN IF NOT EXISTS reorder_point DECIMAL(15,3) DEFAULT 0
        COMMENT '재주문 시점 (S_REORD)',
    ADD COLUMN IF NOT EXISTS supplier_default_id CHAR(36) NULL
        COMMENT '기본 매입처 (S_VENDOR FK)';

CREATE INDEX IF NOT EXISTS idx_items_supplier
    ON items (tenant_id, supplier_default_id);
```
