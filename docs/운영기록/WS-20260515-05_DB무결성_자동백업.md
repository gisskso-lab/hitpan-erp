## 1. 목적

통신 사고 + 디스크 손상 + 랜섬웨어 등 어떤 상황에서도 데이터 손실 0. 헌법 #20 (워크플로우 끊김 금지) + 헌법 #22 (본사 데이터 최소주의) 동시 만족.

## 2. 3단계 백업 정공법

```
① 핫 백업 (3분 주기)
   MariaDB binlog → C:\HitPanBackup\hot\binlog-YYYYMMDDHHMI.bin
   ↓
② 콜드 백업 (1시간 주기)
   mysqldump --single-transaction --routines --triggers
   → C:\HitPanBackup\cold\dump-YYYYMMDDHHMI.sql.gz (gzip 압축)
   ↓
③ 격리 백업 (24시간 주기)
   외장 디스크 또는 NAS (사장님 결재 후 경로 등록)
   E2E 암호화 (헌법 #22) — 본사가 풀 수 없는 형식
```

### 3-2. 핫 백업 워치독 (백엔드매니저)

```csharp
// HitPanWatchdog.exe — 백업 워커 추가
public sealed class BackupWorker : BackgroundService {
    protected override async Task ExecuteAsync(CancellationToken ct) {
        while (!ct.IsCancellationRequested) {
            try {
                await FlushBinlogAsync(ct);
                await CopyBinlogAsync(ct);
            } catch (Exception ex) {
                _logger.LogError(ex, "[Backup] 핫 백업 실패");
            }
            await Task.Delay(TimeSpan.FromMinutes(3), ct);
        }
    }

    private async Task FlushBinlogAsync(CancellationToken ct) {
        using var conn = new MySqlConnection(_connStr);
        await conn.OpenAsync(ct);
        using var cmd = new MySqlCommand("FLUSH BINARY LOGS", conn);
        await cmd.ExecuteNonQueryAsync(ct);
    }

    private async Task CopyBinlogAsync(CancellationToken ct) {
        var src = @"C:\ProgramData\MySQL\MariaDB 11.4\data";
        var dst = @"C:\HitPanBackup\hot";
        Directory.CreateDirectory(dst);
        foreach (var file in Directory.GetFiles(src, "mysql-bin.*")
                                     .Where(f => !f.EndsWith(".index"))) {
            var name = Path.GetFileName(file);
            var dstFile = Path.Combine(dst, $"{name}-{DateTime.Now:yyyyMMddHHmm}");
            if (!File.Exists(dstFile)) {
                File.Copy(file, dstFile);
            }
        }
        // 7일 이상 된 백업 자동 삭제
        foreach (var old in Directory.GetFiles(dst)
                                    .Where(f => File.GetCreationTime(f) < DateTime.Now.AddDays(-7))) {
            File.Delete(old);
        }
    }
}
```

### 3-3. 콜드 백업 (1시간 주기, mysqldump)

```csharp
public sealed class ColdBackupWorker : BackgroundService {
    protected override async Task ExecuteAsync(CancellationToken ct) {
        while (!ct.IsCancellationRequested) {
            await DumpAndCompressAsync(ct);
            await Task.Delay(TimeSpan.FromHours(1), ct);
        }
    }

    private async Task DumpAndCompressAsync(CancellationToken ct) {
        var ts = DateTime.Now.ToString("yyyyMMddHHmm");
        var dumpFile = $@"C:\HitPanBackup\cold\dump-{ts}.sql";
        var gzFile = $"{dumpFile}.gz";

        // mysqldump 실행 — DPAPI 복호화된 비밀번호 사용 (헌법 #5)
        var pwd = DpapiHelper.Unprotect(_encryptedPwd);
        var psi = new ProcessStartInfo(@"C:\Program Files\MariaDB 11.4\bin\mysqldump.exe",
            $"-uhitpan -p{pwd} --single-transaction --routines --triggers " +
            $"--databases hitpan_erp --result-file=\"{dumpFile}\"") {
            UseShellExecute = false,
            RedirectStandardOutput = true,
            RedirectStandardError = true,
        };
        using var proc = Process.Start(psi)!;
        await proc.WaitForExitAsync(ct);
        if (proc.ExitCode != 0) {
            _logger.LogError("[Backup] mysqldump 실패: {Err}",
                await proc.StandardError.ReadToEndAsync());
            return;
        }

        // gzip 압축
        await using (var fs = File.OpenRead(dumpFile))
        await using (var gz = File.Create(gzFile))
        await using (var gzStream = new GZipStream(gz, CompressionLevel.Optimal)) {
            await fs.CopyToAsync(gzStream, ct);
        }
        File.Delete(dumpFile);  // 평문 SQL 삭제

        // 30일 이상 자동 삭제
        foreach (var old in Directory.GetFiles(@"C:\HitPanBackup\cold")
                                    .Where(f => File.GetCreationTime(f) < DateTime.Now.AddDays(-30))) {
            File.Delete(old);
        }
    }
}
```

### 3-4. 격리 백업 (24시간 주기, E2E 암호화)

```csharp
public sealed class IsolatedBackupWorker : BackgroundService {
    protected override async Task ExecuteAsync(CancellationToken ct) {
        while (!ct.IsCancellationRequested) {
            await DelayUntilTimeAsync(TimeSpan.FromHours(3), ct);  // 매일 03:00
            await EncryptAndCopyAsync(ct);
        }
    }

    private async Task EncryptAndCopyAsync(CancellationToken ct) {
        var latestCold = Directory.GetFiles(@"C:\HitPanBackup\cold", "*.sql.gz")
                                  .OrderByDescending(f => f).FirstOrDefault();
        if (latestCold == null) return;

        // AES-256-GCM E2E 암호화
        // 키 = 사장님이 설치 시 등록한 격리 백업 마스터 키 (DPAPI 보호)
        var masterKey = DpapiHelper.Unprotect(_isolatedBackupKey);
        var encrypted = await EncryptFileAsync(latestCold, masterKey, ct);

        // 격리 경로 복사 (사장님 설치 시 등록한 외장 디스크/NAS)
        var dst = Path.Combine(_isolatedPath, Path.GetFileName(latestCold) + ".enc");
        await File.WriteAllBytesAsync(dst, encrypted, ct);
        _logger.LogInformation("[Backup] 격리 백업 완료: {File}", dst);
    }
}
```
