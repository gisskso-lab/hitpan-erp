### 축 4. 무결성 (FK + 워크플로우 + PII)

```sql
-- FK 검증
SELECT 'stock_ledger.item_id orphan' AS check_name, COUNT(*) AS orphan
FROM stock_ledger sl LEFT JOIN items i ON sl.item_id=i.item_id AND sl.tenant_id=i.tenant_id
WHERE sl.tenant_id = @tenant AND i.item_id IS NULL
UNION ALL SELECT 'stock_ledger.warehouse_id orphan', COUNT(*)
FROM stock_ledger sl LEFT JOIN warehouses w ON sl.warehouse_id=w.warehouse_id AND sl.tenant_id=w.tenant_id
WHERE sl.tenant_id = @tenant AND w.warehouse_id IS NULL
UNION ALL SELECT 'collections.partner_id orphan', COUNT(*)
FROM collections c LEFT JOIN partners p ON c.partner_id=p.partner_id AND c.tenant_id=p.tenant_id
WHERE c.tenant_id = @tenant AND p.partner_id IS NULL
UNION ALL SELECT 'tax_invoice_items orphan', COUNT(*)
FROM tax_invoice_items tii LEFT JOIN tax_invoices ti ON tii.invoice_id=ti.invoice_id
WHERE tii.tenant_id = @tenant AND ti.invoice_id IS NULL;
-- 모두 orphan = 0

-- 워크플로우 ID 연결 (WS-MIG-05 검증 재실행)
-- "00000000" 비율 + 양방향 미연결 0건

-- INSERT ONLY 원장 (헌법 #3)
SELECT 'stock_ledger UPDATE 감지', COUNT(*) FROM stock_ledger
WHERE tenant_id = @tenant AND updated_at IS NOT NULL AND updated_at <> created_at;
-- 0건이어야 PASS

-- PII 암호화 (헌법 #5)
SELECT column_name, column_type FROM information_schema.columns
WHERE table_schema = 'hitpan_erp' AND table_name = 'partners'
  AND column_name LIKE '%jumin%' OR column_name LIKE '%resident%';
-- column_type = blob 또는 binary (AES-256 Value Converter 작동 증거)
-- varchar 평문이면 FAIL
```
