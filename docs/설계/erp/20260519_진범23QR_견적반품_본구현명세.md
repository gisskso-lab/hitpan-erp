### MigrateQuotationsAsync 본구현 명세

```csharp
private async Task<int> MigrateQuotationsAsync(
    OleDbConnection oleConn, string tenantId, DateTime now,
    Dictionary<int, string> partnerMap, Dictionary<string, string> itemMap,
    IDbTransaction tx, CancellationToken ct)
{
    // 사장님 자문 #1 결재 후 정확한 테이블/컬럼 확정
    var table = FindMdbTable(oleConn, "DOCFE", "DOCFD", "DOCFC", "DOCFG", "QUOT");
    if (table is null)
    {
        _logger.LogWarning("[MDB마이그레이션] 진범 #23-Q: 견적서 테이블 없음");
        return 0;
    }

    var dt = ReadMdbTable(oleConn, $"SELECT * FROM {table} ORDER BY KE_DT, KE_SEQ");
    if (dt.Rows.Count == 0) return 0;

    // 그룹화 (DOCFE 행 1개 = 라인 1개 추정 → 헤더 그룹화 필요)
    var groups = dt.AsEnumerable().GroupBy(r => new
    {
        Dt = GetStr(r, "KE_DT"),
        Seq = GetInt(r, "KE_SEQ"),
        Buy = GetInt(r, "KE_BUYC"),
    }).ToList();

    var headers = new List<QuotationHeaderRow>();
    var items = new List<QuotationItemRow>();
    int skipPartner = 0, skipItem = 0;

    foreach (var g in groups)
    {
        if (string.IsNullOrWhiteSpace(g.Key.Dt) || g.Key.Dt == "00000000") continue;
        if (!partnerMap.TryGetValue(g.Key.Buy, out var partnerId))
        {
            skipPartner++;
            continue;
        }

        var quoteDate = ParseLegacyDate(g.Key.Dt) ?? now;
        var sourceId = $"mig-docfe-{g.Key.Dt}-{g.Key.Seq}-{g.Key.Buy}";
        var quoteId = Guid.NewGuid().ToString();
        var quoteNo = $"DOCFE-{g.Key.Dt}-{g.Key.Seq:D4}-{g.Key.Buy}";  // 진범 #21 정합

        decimal supplyTotal = 0, vatTotal = 0;
        foreach (var r in g)
        {
            supplyTotal += GetDec(r, "KE_AMT");
            vatTotal += GetDec(r, "KE_VAT");
        }

        headers.Add(new QuotationHeaderRow
        {
            QuoteId = quoteId, TenantId = tenantId, QuoteNo = quoteNo,
            PartnerId = partnerId, QuoteDate = quoteDate,
            Status = "submitted",  // 레거시 견적서는 모두 제출 상태
            TotalAmount = supplyTotal + vatTotal, VatAmount = vatTotal,
            SourceId = sourceId, Now = now,
        });

        foreach (var r in g)
        {
            var pum = GetStr(r, "KE_PUM");
            var ku = GetStr(r, "KE_KU");
            var itemKey = BuildItemKey(pum, ku);  // 진범 #46 정합
            string? itemId = null;
            if (itemMap.TryGetValue(itemKey, out var mappedId)) itemId = mappedId;
            else { skipItem++; itemId = fallbackItemIdLazy; }  // 진범 #34 fallback

            items.Add(new QuotationItemRow
            {
                ItemId = Guid.NewGuid().ToString(), QuoteId = quoteId,
                TenantId = tenantId, MasterItemId = itemId,
                Qty = GetDec(r, "KE_QTY"), UnitPrice = GetDec(r, "KE_DAN"),
                SupplyAmount = GetDec(r, "KE_AMT"), VatAmount = GetDec(r, "KE_VAT"),
            });
        }
    }

    _logger.LogInformation("[MDB마이그레이션] DOCFE 그룹화 — 헤더 {H} 라인 {I} | skip partner {SP} item {SI}",
        headers.Count, items.Count, skipPartner, skipItem);

    if (headers.Count == 0) return 0;

    // BulkCopy 정공법 (sales_deliveries 동형)
    return await BulkCopyQuotationsAsync(mysqlConn, mysqlTx, headers, items, ct);
}
```

### 변경량

- BulkCopyQuotationsAsync (sales_deliveries 동형) 100줄
- QuotationHeaderRow/QuotationItemRow DTO 30줄
- DDL `quotation_items` 테이블 신규 (현재 없음)
